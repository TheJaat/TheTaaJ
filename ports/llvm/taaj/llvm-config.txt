# llvm-config.txt - LLVM's configuration for TheTaaJ: the values CMake
# would have found by testing the system, for tools/build-llvm.py to
# put into include/llvm/Config/*.cmake. NAME=VALUE; a name not here, or
# with nothing after the =, is "no".
#
# The host and target triples and the native target are set by the
# script, from the CPU it builds for.

PACKAGE_NAME=LLVM
PACKAGE_VERSION=23.1.5
PACKAGE_STRING=LLVM 23.1.5
PACKAGE_BUGREPORT=https://github.com/llvm/llvm-project/issues/
PACKAGE_VENDOR=TheTaaJ
LLVM_VERSION_MAJOR=23
LLVM_VERSION_MINOR=1
LLVM_VERSION_PATCH=5

# TheTaaJ has threads now (pthread), but LLVM is still built for one:
# its thread pools and parallel linking are a step of their own, and
# the byte-for-byte checks run one job anyway. LLVM still uses
# std::mutex and its kin.
LLVM_ENABLE_THREADS=0
LLVM_HAS_ATOMICS=1

LLVM_ON_UNIX=1
# Targets.h (llvm-config.h before 19).
LLVM_HAS_X86_TARGET=1
LLVM_HAS_AARCH64_TARGET=1
LLVM_UNREACHABLE_OPTIMIZE=1
LLVM_ENABLE_ABI_BREAKING_CHECKS=0
LLVM_ENABLE_REVERSE_ITERATION=0

# No stack traces on a crash: there is no backtrace() and no way to
# find a function's name in a stripped program.
ENABLE_BACKTRACES=0
ENABLE_CRASH_OVERRIDES=0
LLVM_ENABLE_CRASH_DUMPS=0

# What the C library has.
HAVE_ERRNO_H=1
HAVE_FCNTL_H=1
HAVE_ISATTY=1
HAVE_PTHREAD_H=1
HAVE_PTHREAD_MUTEX_LOCK=1
HAVE_SIGNAL_H=1
HAVE_SYSCONF=1
HAVE_GETPAGESIZE=1
HAVE_SYS_MMAN_H=1
HAVE_SYS_STAT_H=1
HAVE_SYS_RESOURCE_H=1
HAVE_SYS_TIME_H=1
HAVE_SYS_TYPES_H=1
HAVE_UNISTD_H=1
HAVE_SYSEXITS_H=1
HAVE_STRUCT_STAT_ST_MTIM_TV_NSEC=1

# A program starts another with posix_spawn and waits with wait4
# (libc/src/spawn.rs), as clang starts ld.lld. There is no fork: with
# HAVE_POSIX_SPAWN LLVM does not try it.
HAVE_POSIX_SPAWN=1

LLVM_ENABLE_ZLIB=0
LLVM_ENABLE_ZSTD=0
LLVM_ENABLE_DIA_SDK=0
LLVM_USE_INTEL_JITEVENTS=0
LLVM_USE_OPROFILE=0
LLVM_USE_PERF=0
LLVM_FORCE_ENABLE_STATS=0
LLVM_VERSION_PRINTER_SHOW_HOST_TARGET_INFO=1
LLVM_GISEL_COV_ENABLED=0
LLVM_SUPPORT_XCODE_SIGNPOSTS=0
LLVM_WINDOWS_PREFER_FORWARD_SLASH=0

# New in 19-23, all off: ICU and iconv (text in other encodings),
# telemetry, the on-disk content-addressed store, the I/O sandbox, the
# debug-location coverage tracking, and printing the build's options.
HAVE_ICU=0
HAVE_WINDOWS_ICU=0
HAVE_ICONV=0
LLVM_ENABLE_TELEMETRY=0
LLVM_ENABLE_ONDISK_CAS=0
LLVM_ENABLE_IO_SANDBOX=0
LLVM_ENABLE_DEBUGLOC_TRACKING_COVERAGE=0
LLVM_ENABLE_DEBUGLOC_TRACKING_ORIGIN=0
LLVM_VERSION_PRINTER_SHOW_BUILD_CONFIG=0
